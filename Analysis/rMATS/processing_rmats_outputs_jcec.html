#This function sums the comma-separated read counts in the IJC and SJC columns, averages them across replicates, and filters events based on a specified read coverage threshold. It adds two new columns to the dataframe for the average counts in each sample group before applying the filter.

rmats_filter_readCov <- function(df, readCov = 10) {
  
  #Sum comma-separated counts
  sum_counts <- function(x) {
    sapply(strsplit(as.character(x), ","), function(v) {
      sum(as.numeric(v), na.rm = TRUE)
    })
  }
  
  #Sum read counts for coverage filter
  ijc1 <- sum_counts(df$IJC_SAMPLE_1)
  sjc1 <- sum_counts(df$SJC_SAMPLE_1)
  ijc2 <- sum_counts(df$IJC_SAMPLE_2)
  sjc2 <- sum_counts(df$SJC_SAMPLE_2)

  #Number of replicates
  n1 <- 3
  n2 <- 3
  
  #Average coverage per group
  avgCount1 <- (ijc1 + sjc1) / n1
  avgCount2 <- (ijc2 + sjc2) / n2
  
  #Add to dataframe \
  df$averageCountSample1 <- avgCount1
  df$averageCountSample2 <- avgCount2
  
  # Filter like rMATS tutorial
  df_filtered <- df[avgCount1 >= readCov & avgCount2 >= readCov, ]
  
  return(df_filtered)
}

           filter(0.05<IncLevel1 & IncLevel2<0.95) %>%

Volcano Plot Vizualizations

for (i in names){

}
rmats_volcano_gg <- function(df, fdr_cut = 0.01, dpsi_cut = 0.05,
                             label_col = "geneSymbol", top_n_labels = 20, title = NULL) {
  
  df <- df[!is.na(df$FDR) & df$FDR > 0, ]
  df$negLogFDR <- -log10(df$FDR)
  df$negLogFDR[df$negLogFDR > 15] <- 15
  
  df$group <- "Not significant"
  df$group[df$FDR <= fdr_cut & df$IncLevelDifference >= dpsi_cut] <- "Higher in KS1 Hippocampus"
  df$group[df$FDR <= fdr_cut & df$IncLevelDifference <= -dpsi_cut] <- "Higher in WT"
  
  sig <- df[df$group != "Not significant", ]
  sig$rankScore <- sig$negLogFDR * abs(sig$IncLevelDifference)
  sig <- sig[order(-sig$rankScore), ]
  sig_top <- head(sig, top_n_labels)
  
  ggplot(df, aes(IncLevelDifference, negLogFDR, color = group)) +
    geom_point( alpha = 0.7) +
    geom_vline(xintercept = c(-dpsi_cut, dpsi_cut), linetype = "dashed") +
    geom_hline(yintercept = -log10(fdr_cut), linetype = "dashed") +
    geom_text_repel(
      data = sig_top,
      aes(label = .data[[label_col]]),
      size = 3
    )+
    scale_color_manual(values = c(
      "Not significant" = "grey70",
      "Higher in KS1 Hippocampus" = "red",
      "Higher in WT" = "steelblue"
    )) +
    theme_classic(base_size = 14)+
    xlim(-1, 1)+
    ylim(0, 15)+
    labs(x = "Delta PSI",
         y = "-log10(FDR)",
         color = "",
         title = title)
}
